cmake_minimum_required(VERSION 3.20)
project(autocut_color_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(autocut_color_core color_engine.cpp)
target_include_directories(autocut_color_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(autocut_color_core_tests tests/test_color_engine.cpp)
target_link_libraries(autocut_color_core_tests PRIVATE autocut_color_core)
enable_testing()
add_test(NAME color_core COMMAND autocut_color_core_tests)
